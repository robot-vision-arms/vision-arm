cmake_minimum_required(VERSION 3.16)
project(team2_move_group)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pal_statistics REQUIRED)
find_package(moveit_ros_move_group REQUIRED)
# 실행 파일 내부 초기화 구현은 이 버전의 API에 맞춘다. 배포판 라이브러리 변경 시 재검토한다.
find_package(moveit_ros_planning 2.12.4 EXACT REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-sim8 REQUIRED COMPONENTS gui)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(rviz_common REQUIRED)
add_executable(team2_move_group src/team2_move_group.cpp src/team2_scene_monitor.cpp)
# 팀원 2 실행 파일의 초기화 구현만 사용하며 /opt/ros의 공용 라이브러리는 수정하지 않는다.
target_link_options(team2_move_group PRIVATE "-Wl,--export-dynamic")
target_compile_features(team2_move_group PRIVATE cxx_std_17)
target_link_libraries(team2_move_group moveit_ros_move_group::moveit_move_group_capabilities_base)
install(TARGETS team2_move_group DESTINATION lib/${PROJECT_NAME})
add_executable(team2_gz_server src/team2_gz_server.cpp)
target_compile_features(team2_gz_server PRIVATE cxx_std_17)
target_link_libraries(team2_gz_server PRIVATE gz-sim8::gz-sim8 rclcpp::rclcpp)
target_include_directories(team2_gz_server PRIVATE ${pal_statistics_INCLUDE_DIRS})
target_link_libraries(team2_gz_server PRIVATE ${pal_statistics_LIBRARIES})
install(TARGETS team2_gz_server DESTINATION lib/${PROJECT_NAME})
add_executable(team2_gz_gui src/team2_gz_gui.cpp)
target_compile_features(team2_gz_gui PRIVATE cxx_std_17)
target_link_libraries(team2_gz_gui PRIVATE gz-sim8::gz-sim8-gui Qt5::Core)
install(TARGETS team2_gz_gui DESTINATION lib/${PROJECT_NAME})
add_executable(team2_rviz src/team2_rviz.cpp)
target_compile_features(team2_rviz PRIVATE cxx_std_17)
target_link_libraries(team2_rviz PRIVATE rviz_common::rviz_common rclcpp::rclcpp)
install(TARGETS team2_rviz DESTINATION lib/${PROJECT_NAME})
install(FILES src/team2_move_group.cpp src/team2_scene_monitor.cpp DESTINATION share/${PROJECT_NAME}/upstream)
ament_package()
